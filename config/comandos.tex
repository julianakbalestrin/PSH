\newcommand{\atividade}[1]{Atividade~#1}

% ---------------------------------------------------------------------
% Ambiente fichaatividade  -- ficha-síntese de cada atividade
%
%   \begin{fichaatividade}{Título do quadro}{rótulo}
%     \campo{Campo}{Conteúdo}
%     ...
%   \end{fichaatividade}
%
% Campos padronizados (usar sempre esta ordem):
%   Código | Tipo de atividade | Forma de execução | Fonte de recursos |
%   Responsável | Dependências institucionais | Prazo estimado |
%   Produto esperado | Unidade de medida | Evidência de comprovação |
%   Periodicidade | Validação
% ---------------------------------------------------------------------
\newenvironment{fichaatividade}[2]{%
  \quadro[htbp]% (sem \begin, para o tabularx achar \end{fichaatividade})
  \caption{#1}\label{#2}
  \small
  \renewcommand{\arraystretch}{1.25}
  \tabularx{\textwidth}{|>{\columncolor{azulclaro}\bfseries\raggedright\arraybackslash}p{0.32\textwidth}|X|}
  \hline
}{%
  \endtabularx
  \fonte{Adaptado de \citeonline{mop2026}.}
  \endquadro
}
\newcommand{\campo}[2]{#1 & #2 \\ \hline}
